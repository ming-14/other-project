## 进程操作大法

### 结束进程

#### 任务管理器法

如果显示拒绝访问，运行：

```reg
Windows Registry Editor Version 5.00

[HKEY_LOCAL_MACHINE\SOFTWARE\WOW6432Node\TopDomain\e-Learning Class\Student]
"PrenentKill"=hex:00
; 极域好像把 preventKill 打成 PrenentKill 了，笑死我了
```

重启极域即可（注销账户）

#### `Taskkill`法

点击 `Win + R`，输入`taskkill /f /im studentmain.exe`，点击确定

![taskkill](img\taskkill.png)

#### 其它软件结束进程

`Nstd`，`Process Explorer`结束进程皆可

#### 卡死退出法

对于低配置的电脑，在教师发起屏幕广播时，点击开启屏幕录制，并疯狂点击翻转屏幕，有概率卡退

#### 关机（假）退出法

新建一个 Txt 或 Word 文件，输入一点内容。然后注销（或者关机）

此时 Windows 会陆续关闭进程，但是由于 Word 会阻止关机，所有界面会停住，此时退出即可

#### 镜像劫持法

```reg
Windows Registry Editor Version 5.00

[HKEY_LOCAL_MACHINE\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\StudentMain.exe]
"Debugger"="cmd.exe"
```

然后注销电脑

### 切换用户法

首先使用`Dism++`，**火绒安全分析工具**等工具清除可疑的自启项，然后新建用户：

```bat
@echo off
set "username=rikka"
set "password=rikka"

net user %username% %password% /add
net localgroup Administrators %username% /add
```

```vbscript
Set objShell = CreateObject("WScript.Shell")
username = "rikka"
password = "rikka"

' 创建新用户
Set objNetwork = CreateObject("WScript.Network")
Set objUser = GetObject("WinNT://" & objNetwork.ComputerName)
Set objNewUser = objUser.Create("user", username)
objNewUser.SetPassword password
objNewUser.SetInfo

' 添加到管理员组
Set objGroup = GetObject("WinNT://./Administrators")
objGroup.Add("WinNT://" & username)
```

最后注销重新登录即可

### 注册表被禁用

#### VBScript

保存为`.VBS`运行

```vbscript
Set WshShell = WScript.CreateObject("WScript.Shell")

' 删除当前用户的注册表禁用项
On Error Resume Next
WshShell.RegDelete "HKCU\Software\Microsoft\Windows\CurrentVersion\Policies\System\DisableRegistryTools"
If Err.Number <> 0 Then
    WScript.Echo "HKCU 删除失败"
Else
    WScript.Echo "HKCU 注册表解锁成功"
End If
On Error GoTo 0
```

#### HTA

保存为`.HTA`运行

```htaccess
<!DOCTYPE html><html><head><title>注册表解锁工具</title><meta http-equiv="Content-Type" content="text/html; charset=utf-8">
<HTA:APPLICATION 
  ID="RegistryUnlocker"
  APPLICATIONNAME="Registry Unlock Tool"
  BORDER="none"
  CAPTION="no"
  SHOWINTASKBAR="no"
  WINDOWSTATE="minimize"
>
<script language="VBScript">
Sub Window_OnLoad
  Set WshShell = CreateObject("WScript.Shell")
  On Error Resume Next
  WshShell.RegDelete "HKCU\Software\Microsoft\Windows\CurrentVersion\Policies\System\DisableRegistryTools"
  If Err.Number = 0 Then
    MsgBox "HKCU 注册表解锁成功", 64, "操作完成"
  Else
    MsgBox "HKCU 删除失败（可能不存在或权限不足）", 48, "操作结果"
  End If
  window.Close
End Sub
</script></head><body></body></html>
```

#### JavaScript

保存为`.JS`运行

```javascript
var WshShell = WScript.CreateObject("WScript.Shell");
try {
    WshShell.RegDelete("HKCU\\Software\\Microsoft\\Windows\\CurrentVersion\\Policies\\System\\DisableRegistryTools");
    WScript.Echo("HKCU 注册表解锁成功");
} catch (e) {
    WScript.Echo("HKCU 删除失败");
}
```

#### PowerShell

保存为`.PS1`运行

```powershell
try {
    Remove-ItemProperty -Path "HKCU:\Software\Microsoft\Windows\CurrentVersion\Policies\System" -Name "DisableRegistryTools" -ErrorAction Stop
    Write-Host "HKCU 注册表解锁成功"
} catch {
    Write-Host "HKCU 删除失败"
}
```

#include <windows.h>
#include <tlhelp32.h>
#include <wintrust.h>
#include <softpub.h>
#include <psapi.h>
#include <shlwapi.h>
#include <string>
#include <vector>
#include <iostream>
#include <algorithm>
#include <fstream>

#define NT_SUCCESS(Status) (((NTSTATUS)(Status)) >= 0)

#pragma comment(lib, "wintrust.lib")
#pragma comment(lib, "shlwapi.lib")

// 声明未文档化的NTAPI函数
typedef NTSTATUS(NTAPI* pfnNtSetInformationProcess)(
    HANDLE ProcessHandle,
    ULONG ProcessInformationClass,
    PVOID ProcessInformation,
    ULONG ProcessInformationLength);

#define ProcessBreakOnTermination 29

// 全局日志文件流对象
std::ofstream logFile;

// 验证文件数字签名
bool HasValidSignature(const char* filePath) {
    WINTRUST_FILE_INFO fileInfo = { 0 };
    fileInfo.cbStruct = sizeof(WINTRUST_FILE_INFO);

    // 转换多字节路径为宽字符
    wchar_t wFilePath[MAX_PATH];
    MultiByteToWideChar(CP_ACP, 0, filePath, -1, wFilePath, MAX_PATH);
    fileInfo.pcwszFilePath = wFilePath;

    WINTRUST_DATA trustData = { 0 };
    trustData.cbStruct = sizeof(WINTRUST_DATA);
    trustData.dwUIChoice = WTD_UI_NONE;
    trustData.fdwRevocationChecks = WTD_REVOKE_NONE;
    trustData.dwUnionChoice = WTD_CHOICE_FILE;
    trustData.pFile = &fileInfo;
    trustData.dwStateAction = WTD_STATEACTION_VERIFY;
    trustData.dwProvFlags = WTD_SAFER_FLAG;

    GUID policyGuid = WINTRUST_ACTION_GENERIC_VERIFY_V2;
    LONG status = WinVerifyTrust(NULL, &policyGuid, &trustData);

    trustData.dwStateAction = WTD_STATEACTION_CLOSE;
    WinVerifyTrust(NULL, &policyGuid, &trustData);

    return status == ERROR_SUCCESS;
}

// 获取进程完整路径
std::string GetProcessPath(DWORD processId) {
    char path[MAX_PATH] = { 0 };
    HANDLE hProcess = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, FALSE, processId);
    if (hProcess) {
        DWORD size = MAX_PATH;
        if (QueryFullProcessImageNameA(hProcess, 0, path, &size)) {
            CloseHandle(hProcess);
            return path;
        }
        CloseHandle(hProcess);
    }
    return "";
}

// 移除关键进程标志
bool RemoveCriticalProcessFlag(HANDLE hProcess) {
    HMODULE hNtdll = GetModuleHandleA("ntdll.dll");
    if (!hNtdll) return false;

    pfnNtSetInformationProcess NtSetInformationProcess =
        (pfnNtSetInformationProcess)GetProcAddress(hNtdll, "NtSetInformationProcess");
    if (!NtSetInformationProcess) return false;

    ULONG BreakOnTermination = 0;
    NTSTATUS status = NtSetInformationProcess(hProcess,
        ProcessBreakOnTermination,
        &BreakOnTermination,
        sizeof(ULONG));

    return NT_SUCCESS(status);
}

// 终止单个进程
void TerminateProcessById(DWORD pid, DWORD selfPid) {
    if (pid == selfPid || pid == 0) return;

    HANDLE hProcess = OpenProcess(PROCESS_TERMINATE | PROCESS_SET_INFORMATION, FALSE, pid);
    if (!hProcess) return;

    // 尝试移除关键进程标志
    RemoveCriticalProcessFlag(hProcess);

    // 终止进程
    TerminateProcess(hProcess, 0);
    CloseHandle(hProcess);
}

// 停止无签名服务
void StopUnsignedServices(DWORD selfPid) {
    SC_HANDLE scm = OpenSCManager(NULL, NULL, SC_MANAGER_ENUMERATE_SERVICE | SC_MANAGER_CONNECT);
    if (!scm) return;

    DWORD servicesSize = 0;
    DWORD serviceCount = 0;
    EnumServicesStatusExA(
        scm,
        SC_ENUM_PROCESS_INFO,
        SERVICE_TYPE_ALL,
        SERVICE_STATE_ALL,
        NULL,
        0,
        &servicesSize,
        &serviceCount,
        NULL,
        NULL
    );

    if (servicesSize == 0) {
        CloseServiceHandle(scm);
        return;
    }

    std::vector<BYTE> buffer(servicesSize);
    EnumServicesStatusExA(
        scm,
        SC_ENUM_PROCESS_INFO,
        SERVICE_TYPE_ALL,
        SERVICE_STATE_ALL,
        buffer.data(),
        buffer.size(),
        &servicesSize,
        &serviceCount,
        NULL,
        NULL
    );

    ENUM_SERVICE_STATUS_PROCESSA* services =
        reinterpret_cast<ENUM_SERVICE_STATUS_PROCESSA*>(buffer.data());

    for (DWORD i = 0; i < serviceCount; ++i) {
        SC_HANDLE service = OpenServiceA(
            scm,
            services[i].lpServiceName,
            SERVICE_QUERY_CONFIG | SERVICE_STOP
        );

        if (!service) continue;

        // 获取服务配置
        DWORD needed = 0;
        QueryServiceConfigA(service, NULL, 0, &needed);
        std::vector<BYTE> configBuffer(needed);
        QUERY_SERVICE_CONFIGA* config =
            reinterpret_cast<QUERY_SERVICE_CONFIGA*>(configBuffer.data());

        if (QueryServiceConfigA(service, config, configBuffer.size(), &needed)) {
            // 跳过驱动程序服务
            if (config->dwServiceType != SERVICE_WIN32_OWN_PROCESS) {
                CloseServiceHandle(service);
                continue;
            }

            // 提取可执行文件路径
            std::string binPath = config->lpBinaryPathName;
            if (binPath[0] == '"') {
                binPath = binPath.substr(1, binPath.find('"', 1) - 1);
            }
            else {
                size_t spacePos = binPath.find(' ');
                if (spacePos != std::string::npos) {
                    binPath = binPath.substr(0, spacePos);
                }
            }

            // 检查签名并停止服务
            if (!HasValidSignature(binPath.c_str())) {
                logFile << "Service " << services[i].lpServiceName << " (" << binPath << ") is unsigned, stopping it." << std::endl;
                SERVICE_STATUS status;
                ControlService(service, SERVICE_CONTROL_STOP, &status);
            }
        }
        CloseServiceHandle(service);
    }
    CloseServiceHandle(scm);
}

int main() {
    // 打开日志文件
    logFile.open("./log.log", std::ios::out | std::ios::app);
    if (!logFile.is_open()) {
        std::cerr << "Failed to open log file." << std::endl;
        return 1;
    }

    DWORD selfPid = GetCurrentProcessId();

    // 遍历所有进程
    HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snapshot == INVALID_HANDLE_VALUE) {
        logFile << "Failed to create process snapshot." << std::endl;
        logFile.close();
        return 1;
    }

    PROCESSENTRY32 pe = { 0 };
    pe.dwSize = sizeof(PROCESSENTRY32);

    if (!Process32First(snapshot, &pe)) {
        logFile << "Process32First failed." << std::endl;
        CloseHandle(snapshot);
        logFile.close();
        return 1;
    }

    do {
        if (pe.th32ProcessID == selfPid) continue;

        std::string path = GetProcessPath(pe.th32ProcessID);
        if (path.empty()) continue;

        if (!HasValidSignature(path.c_str())) {
            logFile << "Process " << pe.th32ProcessID << " (" << path << ") is unsigned, terminating it." << std::endl;
            TerminateProcessById(pe.th32ProcessID, selfPid);
        }
    } while (Process32Next(snapshot, &pe));

    CloseHandle(snapshot);

    // 处理服务
    StopUnsignedServices(selfPid);
    logFile.close(); // 关闭日志文件
    system("pause");
    return 0;
}
